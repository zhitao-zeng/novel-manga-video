  --novel-dir outputs/xinghai \
  --state-dir outputs/xinghai/repair_manager/sd_audit \
  --queue outputs/xinghai/repair_manager/sd_audit/shared_audit.sqlite3 \
  --lane qwen --workers 8 --sync-reviews --max-tokens 1800
